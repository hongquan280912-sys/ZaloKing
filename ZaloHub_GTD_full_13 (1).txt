local ownCo = setmetatable({}, { __mode = "k" })
local function ownCall(f)
    local co = coroutine.running()
    ownCo[co] = true
    local r = table.pack(pcall(f))
    ownCo[co] = nil
    return table.unpack(r, 1, r.n)
end

local remoteCache = {}
local function remote(name)
    local r = remoteCache[name]
    if r and r.Parent then return r end
    for _, v in ipairs(RS:GetDescendants()) do
        if v.Name == name and (v:IsA("RemoteEvent") or v:IsA("RemoteFunction")) then
            remoteCache[name] = v
            return v
        end
    end
end

local function fire(name, ...)
    local r = remote(name)
    if not r then return end
    local a = table.pack(...)
    ownCall(function()
        if r:IsA("RemoteFunction") then
            r:InvokeServer(table.unpack(a, 1, a.n))
        else
            r:FireServer(table.unpack(a, 1, a.n))
        end
    end)
end

local function rootPart()
    local ch = LP.Character
    return ch and ch:FindFirstChild("HumanoidRootPart")
end

local function isShown(o)
    while o and o ~= game do
        if o:IsA("GuiObject") and not o.Visible then return false end
        if o:IsA("ScreenGui") and not o.Enabled then return false end
        o = o.Parent
    end
    return true
end

local function press(btn)
    local ok = false
    if firesignal then
        ok = pcall(function() firesignal(btn.MouseButton1Click) end)
        pcall(function() firesignal(btn.Activated) end)
    end
    if not ok and getconnections then
        for _, sig in ipairs({ btn.MouseButton1Click, btn.Activated }) do
            pcall(function()
                for _, c in ipairs(getconnections(sig)) do c:Fire() end
            end)
        end
    end
end

local function takeSnapshot()
    local s = { wave = nil, present = false, btn = {} }
    local pg = LP:FindFirstChild("PlayerGui")
    if not pg then return s end
    local lvl = S.voteLevel > 0 and ("level " .. S.voteLevel) or nil
    for _, o in ipairs(pg:GetDescendants()) do
        if o:IsA("TextLabel") or o:IsA("TextButton") then
            local t = o.Text
            if t ~= "" and #t < 40 then
                local lt = t:lower()
                if WANT[lt] or lt == lvl then
                    if isShown(o) and not s.btn[lt] then s.btn[lt] = o end
                elseif lt:find("wave", 1, true) then
                    local w = t:match("Wave%s*(%d+)%s*/%s*%d+")
                    if (w or lt:find("next wave", 1, true)) and isShown(o) then
                        s.present = true
                        if w then s.wave = tonumber(w) end
                    end
                elseif lt:find("enemies arrive", 1, true) or lt:find("enemies incoming", 1, true) then
                    if isShown(o) then s.present = true end
                end
            end
        end
    end
    return s
end

    local pg = LP:FindFirstChild("PlayerGui")

    while b and b ~= pg and not b:IsA("GuiButton") do b = b.Parent end
    if b and b:IsA("GuiButton") then

end

-- Theo dõi wave hiện tại và phát hiện trận mới
local function updateWave(s)
    if s.present then
        if goneFlag then
            matchSerial += 1
            curWave = s.wave or 0
            waveStart = os.clock()
            goneFlag = false
        elseif s.wave then
            if s.wave < curWave then
                matchSerial += 1
                curWave = s.wave
                waveStart = os.clock()
            elseif s.wave ~= curWave then
                curWave = s.wave
                waveStart = os.clock()
            end
        end
        lastPresent = os.clock()
    elseif os.clock() - lastPresent > 6 then
        goneFlag = true
    end
end

local function describe(v, d)
    d = d or 0
    local t = typeof(v)
    if t == "Instance" then return v.ClassName .. ":" .. v.Name end
    if t == "Vector3" then return ("V3(%.0f,%.0f,%.0f)"):format(v.X, v.Y, v.Z) end
    if t == "CFrame" then
        local p = v.Position
        return ("CF(%.0f,%.0f,%.0f)"):format(p.X, p.Y, p.Z)
    end
    if t == "string" then return '"' .. v:sub(1, 24) .. '"' end
    if t == "table" then
        if d >= 2 then return "{...}" end
        local out, n = {}, 0
        for k, x in pairs(v) do
            if k ~= "n" then
                n += 1
                if n > 6 then
                    out[#out + 1] = "..."
                    break
                end
                out[#out + 1] = tostring(k) .. "=" .. describe(x, d + 1)
            end
        end
        return "{" .. table.concat(out, ", ") .. "}"
    end
    return tostring(v)
end

local function describeArgs(a)
    local out = {}
    for i = 1, (a.n or #a) do out[#out + 1] = describe(a[i]) end
    return table.concat(out, ", ")
end

local SKIPFOLDER = { towers = true, units = true, tower = true, unit = true, map = true, placed = true }

local function underSkipFolder(m)
    local p = m.Parent

        if SKIPFOLDER[p.Name:lower()] then return true end

local function modelOf(b)
    local a = b.Adornee or b.Parent
    if not a then return nil end
    if a:IsA("Model") then return a end
    return a:FindFirstAncestorOfClass("Model")
end

local function mobAlive(m)
    if not (m and m.Parent) then return false end
    local h = m:FindFirstChildOfClass("Humanoid")
    if h and h.Health <= 0 then return false end
    return true
end

-- "Workspace.Contexts.xxx.Entities" -> Instance
local function resolvePath(s)
    if not s:find(".", 1, true) then return nil end
    local cur = game
    for part in s:gmatch("[^%.]+") do
        cur = cur and cur:FindFirstChild(part)
        if not cur then return nil end
    end
    return cur
end

local enemyFolder

    if S.enemyName ~= "" then
        local f = resolvePath(S.enemyName) or workspace:FindFirstChild(S.enemyName, true)
        if f then return f end
    end
    if enemyFolder and enemyFolder.Parent then return enemyFolder end
    for _, n in ipairs({ "Enemies", "Enemy", "Mobs", "Mob", "Monsters", "Zombies" }) do
        local f = workspace:FindFirstChild(n)

end

-- Trong thư mục Entities quái có tên dạng enemy_aphid, enemy_ant...
local function enemyChildren(folder)
    local named, all = {}, {}
    for _, m in ipairs(folder:GetChildren()) do
        local isNamed = m.Name:lower():sub(1, 5) == "enemy"
        if (m:IsA("Model") or m:IsA("BasePart")) and (isNamed or m:IsA("Model")) then
            all[#all + 1] = m
            if isNamed then named[#named + 1] = m end
        end
    end
    return all
end

local heavyCache, heavyHow, heavyT = {}, "", -10

local function getEnemies()
    local folder = findEnemyFolder()
    if folder then
        return enemyChildren(folder), "thư mục " .. folder.Name
    end

    -- Không thấy thư mục: quét nặng, tối đa 4 giây một lần
    if os.clock() - heavyT < 4 then
        local keep = {}
        for _, m in ipairs(heavyCache) do
            if m.Parent then keep[#keep + 1] = m end
        end
        return keep, heavyHow
    end
    heavyT = os.clock()

    local out = {}
    local set = {}
    for _, container in ipairs({ workspace, LP:FindFirstChild("PlayerGui") }) do
        if container then
            for _, b in ipairs(container:GetDescendants()) do
                if b:IsA("BillboardGui") then
                    local hasHp = false
                    for _, t in ipairs(b:GetDescendants()) do
                        if t:IsA("TextLabel") and t.Text:match("^%s*%d+%s*/%s*%d+%s*$") then
                            hasHp = true
                            break
                        end
                    end
                    if hasHp then
                        local m = modelOf(b)
                        if m and m ~= LP.Character and not Players:GetPlayerFromCharacter(m)
                            and not underSkipFolder(m) and not set[m] then
                            set[m] = true
                            out[#out + 1] = m
                        end
                    end
                end
            end
        end
    end
    if #out > 0 then
        heavyCache, heavyHow = out, "thanh máu x/y"
        return out, heavyHow
    end

    for _, h in ipairs(workspace:GetDescendants()) do
        if h:IsA("Humanoid") and h.Health > 0 then
            local m = h.Parent
            if m and m:IsA("Model") and m ~= LP.Character
                and not Players:GetPlayerFromCharacter(m) and not underSkipFolder(m) then
                out[#out + 1] = m
            end
        end
    end
    heavyCache, heavyHow = out, "Humanoid"
    return out, heavyHow
end

local function posOf(m)
    local ok, p = pcall(function()
        if m:IsA("Model") then return m:GetPivot().Position end
        return m.Position
    end)
    if ok then return p end
end

local function nearestEnemy()
    local root = rootPart()
    if not root then return nil end
    local list = getEnemies()
    local best, bd
    for _, m in ipairs(list) do
        if mobAlive(m) then
            local p = posOf(m)
            if p then
                local d = (p - root.Position).Magnitude
                if not bd or d < bd then best, bd = m, d end
            end
        end
    end
    return best
end

-- Con đi đầu = con xuất hiện sớm nhất còn sống (bám tới khi nó chết)
local firstSeen = setmetatable({}, { __mode = "k" })
local followTarget = nil
local function leadEnemy()
    local list = getEnemies()
    local now = os.clock()
    local best, bt
    for _, m in ipairs(list) do
        if mobAlive(m) then
            local ts = firstSeen[m]
            if not ts then
                ts = now
                firstSeen[m] = ts
            end
            if not bt or ts < bt then best, bt = m, ts end
        end
    end
    return best
end

-- ===== LỌC QUÁI THẬT + BÁM QUÁI CHẮC (khối do...end: không thêm local cấp cao) =====
do
    local rawEnemies = getEnemies
    local mv = setmetatable({}, { __mode = "k" })     -- theo dõi di chuyển
    local skipUntil = setmetatable({}, { __mode = "k" })
    local stuck = setmetatable({}, { __mode = "k" })

    local function looksLikeUnit(m)
        for _, k in ipairs({ "Owner", "OwnerId", "OwnerUserId", "UserId", "PlacedBy", "Placer" }) do
            if m:GetAttribute(k) ~= nil then return true end
        end
        if m:FindFirstChild("Owner") or m:FindFirstChild("OwnerId") then return true end
        return false
    end

    -- Quái: tên bắt đầu "enemy" HOẶC đã từng di chuyển. Unit đứng yên nên không lọt vào.
    local function realEnemy(m)
        if m.Name:lower():sub(1, 5) == "enemy" then return true end
        if looksLikeUnit(m) then return false end
        local p = posOf(m)
        if not p then return false end
        local info = mv[m]
        if not info then
            mv[m] = { p0 = p, moved = false }
            return false
        end
        if info.moved then return true end
        if (p - info.p0).Magnitude > 1.5 then
            info.moved = true
            return true
        end
        return false
    end

    getEnemies = function()
        local list, how = rawEnemies()
        local out = {}
        for _, m in ipairs(list) do
            if realEnemy(m) then out[#out + 1] = m end
        end
        return out, how
    end

    local function deadish(m)
        if not mobAlive(m) then return true end
        local hp = m:GetAttribute("Health") or m:GetAttribute("health") or m:GetAttribute("HP")
        if type(hp) == "number" and hp <= 0 then return true end
        return false
    end

    -- cur = mục tiêu đang bám (nếu còn sống và vẫn đang di chuyển thì giữ nguyên)
    leadEnemy = function(cur)
        local now = os.clock()
        if cur and not deadish(cur) then
            local p = posOf(cur)
            local st = stuck[cur]
            if p then
                if not st or (p - st.p).Magnitude > 0.5 then
                    stuck[cur] = { p = p, t = now }
                    return cur
                elseif now - st.t <= 4 then
                    return cur
                end
                skipUntil[cur] = now + 8 -- đứng im quá 4 giây: tạm bỏ qua để bám con khác
            end
        end
        local best, bt, any, at
        for _, m in ipairs(getEnemies()) do
            if not deadish(m) then
                local ts = firstSeen[m]
                if not ts then
                    ts = now
                    firstSeen[m] = ts
                end
                if not at or ts < at then any, at = m, ts end
                if (skipUntil[m] or 0) < now and (not bt or ts < bt) then best, bt = m, ts end
            end
        end
        return best or any
    end
end

local function kindOf(name)
    name = (name or ""):lower()
    if name:find("place", 1, true) then return "place" end
    if name:find("upgrade", 1, true) then return "upgrade" end
    if name:find("sell", 1, true) or name:find("remove", 1, true) or name:find("delete", 1, true)
        or name:find("destroy", 1, true) or name:find("refund", 1, true) or name:find("retire", 1, true) then
        return "sell"
    end
    return "other"
end

-- ===== MACRO: LƯU / NẠP (JSON) =====
local function ser(v)
    local t = typeof(v)
    if t == "number" or t == "string" or t == "boolean" then return v end
    if t == "Vector3" then return { _t = "V3", x = v.X, y = v.Y, z = v.Z } end
    if t == "CFrame" then return { _t = "CF", c = { v:GetComponents() } } end
    if t == "Instance" then return { _t = "I", name = v.Name } end
    if t == "table" then
        local e = {}
        for k, x in pairs(v) do
            local sk, sx = ser(k), ser(x)
            if sk ~= nil and sx ~= nil then e[#e + 1] = { sk, sx } end
        end
        return { _t = "T", e = e }
    end
    return nil
end

local function deser(v)
    if type(v) ~= "table" then return v end
    if v._t == "V3" then return Vector3.new(v.x, v.y, v.z) end
    if v._t == "CF" then return CFrame.new(table.unpack(v.c)) end
    if v._t == "I" then return remote(v.name) end
    if v._t == "T" then
        local o = {}
        for _, kv in ipairs(v.e) do
            local k = deser(kv[1])
            if k ~= nil then o[k] = deser(kv[2]) end
        end
        return o
    end
    return v
end

local function serUnits(list)
    if not list then return nil end
    local out = {}
    for _, x in ipairs(list) do
        out[#out + 1] = { i = x.i, name = x.name, parent = x.parent, pos = { x.pos.X, x.pos.Y, x.pos.Z } }
    end
    return out
end

local function deserUnits(list)
    if type(list) ~= "table" then return nil end
    local out = {}
    for _, x in ipairs(list) do
        if type(x) == "table" and type(x.pos) == "table" and #x.pos >= 3 then
            out[#out + 1] = {
                i = x.i, name = x.name, parent = x.parent,
                pos = Vector3.new(x.pos[1], x.pos[2], x.pos[3]),
            }
        end
    end
    return #out > 0 and out or nil
end

local function v3t(v) return v and { v.X, v.Y, v.Z } or nil end
local function t3v(t)
    return (type(t) == "table" and #t == 3) and Vector3.new(t[1], t[2], t[3]) or nil
end

local function macroToTable()

    for _, e in ipairs(rec) do
        out[#out + 1] = {
            t = e.t, remote = e.name, method = e.method,

            pos = e.pos and { e.pos:GetComponents() } or nil,
            units = serUnits(e.units),
            wave = e.wave, wt = e.wt,
            kind = e.kind, unit = e.unit, at = v3t(e.at),
            fx = e.fx, fy = e.fy,
        }
    end
    return out
end

local setters = {}

local function applyConfig(data)
    for k, v in pairs(data) do
        if S[k] ~= nil and type(S[k]) == type(v) then
            S[k] = v
            if setters[k] then setters[k](v) end
        end
    end
end

local function applyMacro(data)
    local newRec = {}
    for _, d in ipairs(data) do
        local args = deser(d.args)
        if type(args) ~= "table" then args = { n = 0 } end
        args.n = args.n or #args
        local pos
        if type(d.pos) == "table" and #d.pos >= 12 then
            pos = CFrame.new(table.unpack(d.pos))
        end
        newRec[#newRec + 1] = {
            t = d.t, name = d.remote, inst = remote(d.remote),
            method = d.method, args = args, placedId = d.placedId, pos = pos,
            units = deserUnits(d.units), wave = d.wave, wt = d.wt,
            kind = d.kind or kindOf(d.remote), unit = d.unit, at = t3v(d.at),
            fx = d.fx, fy = d.fy,
        }
    end
    rec = newRec
    return #newRec
end

local function importText(raw)
    local ok, obj = pcall(function() return HttpService:JSONDecode(raw) end)
    if not ok or type(obj) ~= "table" then return "Dữ liệu không hợp lệ" end
    if obj.kind == "gtd_config" and type(obj.data) == "table" then
        applyConfig(obj.data)
        return "Đã nạp config"
    elseif obj.kind == "gtd_macro" and type(obj.data) == "table" then
        local n = applyMacro(obj.data)
        saveMacroFile()
        return "Đã nạp macro (" .. n .. " bước)"
    end
    return "Không nhận ra dữ liệu này"
end

local function recToText()
    local out = {}
    for i, e in ipairs(rec) do
        local p = e.pos and e.pos.Position
        out[#out + 1] = ("%d | %.1fs | %s:%s(%s)%s%s%s"):format(
            i, e.t, e.name, e.method, describeArgs(e.args),
            e.placedId ~= nil and (" -> id " .. tostring(e.placedId)) or "",
            p and (" @ " .. ("%.0f,%.0f,%.0f"):format(p.X, p.Y, p.Z)) or "",
            e.wave and (" | wave " .. e.wave .. " +" .. ("%.0f"):format(e.wt or 0) .. "s") or "")
        if e.result ~= nil then
            out[#out] = out[#out] .. " | trả về: " .. describeArgs(type(e.result) == "table" and e.result or { e.result, n = 1 }):sub(1, 160)
        end
        if e.err then out[#out] = out[#out] .. " | LỖI: " .. e.err end
    end
    return table.concat(out, "\n")
end

local seen, seenOrder, lastShown = {}, {}, ""
local allLog = {} -- nhật ký MỌI remote kèm tham số (để chẩn đoán)
local placedInfo = {} -- [id server trả về] = { unit, at }
local SKIP = { "ping", "heartbeat", "camera", "mouse", "cursor", "latency", "fps",
    "analytic", "telemetry", "report", "replicat", "chat", "afk" }

local KEYS = { "place", "upgrade", "sell", "target", "lock", "remove", "delete",
    "destroy", "refund", "retire", "auto", "abilit", "skill", "ultimate" }
local NOTKEYS = { "autoskip", "speed", "tick", "replay", "vote", "afk", "camera", "autoplay" }

local function wanted(name)
    local n = name:lower()
    for _, k in ipairs(NOTKEYS) do
        if n:find(k, 1, true) then return false end
    end
    if S.recAll then
        for _, k in ipairs(SKIP) do
            if n:find(k, 1, true) then return false end
        end
        return true
    end
    for _, k in ipairs(KEYS) do
        if n:find(k, 1, true) then return true end
    end
    return false
end

-- Lấy id unit từ kết quả đặt unit: {true, id} hoặc bảng có Id / UnitId...
local extractId
extractId = function(r)
    if type(r) ~= "table" then return nil end
    -- giá trị trả về chỉ là một số / chuỗi id
    if (type(r[1]) == "number" or type(r[1]) == "string") and r[2] == nil then return r[1] end
    if r[1] == true then
        local v = r[2]
        if type(v) == "number" or type(v) == "string" or typeof(v) == "Instance" then return v end
    end
    for _, k in ipairs({ "Id", "id", "ID", "UnitId", "unitId", "UnitID", "UID", "Uid", "uid",
        "Guid", "GUID", "guid", "Key", "key", "Index", "index", "Slot", "UnitUID", "TowerId", "towerId" }) do
        local v = r[k]
        if type(v) == "number" or type(v) == "string" then return v end
    end
    -- bảng kiểu {true, {Id = ...}} hoặc {{...}}
    for _, v in pairs(r) do
        if type(v) == "table" then
            local x = extractId(v)
            if x ~= nil then return x end
        end
    end
    return nil
end

-- Nhớ vị trí của các unit (model trong workspace) được truyền làm tham số
local function captureUnits(args)
    local list
    for i = 1, (args.n or #args) do
        local v = args[i]
        if typeof(v) == "Instance" and not v:IsA("RemoteEvent") and not v:IsA("RemoteFunction")
            and v:IsDescendantOf(workspace) then
            local ok, p = pcall(function() return v:GetPivot().Position end)
            if ok then
                list = list or {}
                list[#list + 1] = {
                    i = i, name = v.Name,
                    parent = v.Parent and v.Parent.Name or nil, pos = p,
                }
            end
        end
    end
    return list
end

-- tìm tên unit (string) và vị trí đặt (Vector3/CFrame) trong args
local function scanArgs(args)
    local uname, at
    local function look(v)
        local tv = typeof(v)
        if tv == "Vector3" and not at then at = v
        elseif tv == "CFrame" and not at then at = v.Position
        elseif tv == "string" and not uname then uname = v end
    end
    for i = 1, (args.n or #args) do
        local v = args[i]
        if type(v) == "table" then
            for _, x in pairs(v) do look(x) end
        else
            look(v)
        end
    end
    return uname, at
end

local lastPlaced = nil
local function findInfo(args)
    for i = 1, (args.n or #args) do
        local v = args[i]
        local tv = type(v)
        if tv == "number" or tv == "string" then
            local f = placedInfo[tostring(v)]
            if f then return f end
        elseif tv == "table" then
            for _, x in pairs(v) do
                if type(x) == "number" or type(x) == "string" then
                    local f = placedInfo[tostring(x)]
                    if f then return f end
                end
            end
        end
    end
end

local function addRecInner(self, m, args, result)
    local name = self.Name
    if not seen[name] then
        seen[name] = 0
        seenOrder[#seenOrder + 1] = name
    end
    seen[name] += 1
    do
        local ln = name:lower()
        local noise = false
        for _, k in ipairs({ "ping", "heartbeat", "camera", "mouse", "cursor", "latency", "fps" }) do
            if ln:find(k, 1, true) then noise = true break end
        end
        if not noise and #allLog < 400 then
            allLog[#allLog + 1] = ("%.1fs | %s:%s(%s)%s"):format(
                os.clock() - recStart, name, m, describeArgs(args):sub(1, 220),
                type(result) == "table" and (" => " .. describeArgs(result):sub(1, 160)) or "")
        end
    end
    if not wanted(name) then
        if name ~= lastShown then
            lastShown = name
            setStatus("Bỏ qua remote: " .. name)
        end
        return
    end
    if #rec == 0 then recStart = os.clock() end

    local root = rootPart()
    local e = {
        t = os.clock() - recStart, inst = self, name = name, method = m, args = args,
        kind = kindOf(name),
        pos = root and root.CFrame or nil, -- vị trí nhân vật đang đứng
        wave = curWave > 0 and curWave or nil,
        wt = os.clock() - waveStart,
    }
    e.units = captureUnits(args)
    e.result = result

    if e.kind == "place" then
        e.unit, e.at = scanArgs(args)
        if m == "InvokeServer" then
            e.placedId = extractId(result)
            if e.placedId ~= nil and typeof(e.placedId) ~= "Instance" then
                placedInfo[tostring(e.placedId)] = { unit = e.unit, at = e.at }
                lastPlaced = placedInfo[tostring(e.placedId)]
            end
        end
    elseif e.kind == "upgrade" or e.kind == "sell" then
        local info = findInfo(args)
        if info then e.unit, e.at = info.unit, info.at end
        if not e.unit and e.units then
            e.unit, e.at = e.units[1].name, e.units[1].pos
        end
        if not e.unit and not name:lower():find("all", 1, true) then
            e.unlinked = true
            setStatus("⚠ " .. name .. ": chưa nhận ra unit (args: " .. describeArgs(args) .. ")")
        end
    end

    rec[#rec + 1] = e
    lastShown = name
    setStatus("Đã ghi " .. #rec .. " bước · " .. name .. (e.wave and (" · wave " .. e.wave) or ""))
end

-- Bọc addRec: lỗi khi ghi không được làm mất bước hoặc làm hỏng lệnh gốc của game
local getId = getthreadidentity or get_thread_identity or (syn and syn.get_thread_identity)
local setId = setthreadidentity or set_thread_identity or (syn and syn.set_thread_identity)

local function isDup(nm, m, args)
    local last = rec[#rec]
    return last ~= nil and last.name == nm and last.method == m
        and (os.clock() - recStart) - last.t < 0.15
        and tostring(last.args and last.args[1]) == tostring(args and args[1])
end

local function addRecCore(self, m, args, result)
    local okN, nm = pcall(function() return self.Name end)
    if not okN then return end
    -- hai lớp hook có thể cùng bắt một lệnh: bỏ lệnh trùng, nhưng giữ kết quả server trả về
    if isDup(nm, m, args) then
        local last = rec[#rec]
        if last.result == nil and result ~= nil then last.result = result end
        return
    end
    local ok, err = pcall(addRecInner, self, m, args, result)
    if ok then return end
    pcall(function()
        if not wanted(nm) then return end
        if #rec == 0 then recStart = os.clock() end
        local root = rootPart()
        rec[#rec + 1] = {
            t = os.clock() - recStart, inst = self, name = nm, method = m, args = args,
            kind = kindOf(nm), pos = root and root.CFrame or nil,
            wave = curWave > 0 and curWave or nil, wt = os.clock() - waveStart,
            result = result, err = tostring(err):sub(1, 120),
        }
        setStatus("Đã ghi " .. #rec .. " bước · " .. nm .. " (lỗi nhận diện unit: " .. tostring(err):sub(1, 80) .. ")")
    end)
end

-- Lệnh của game có thể đến từ luồng có quyền thấp: nâng quyền tạm thời khi ghi rồi trả về như cũ
local function addRec(self, m, args, result)
    local prev
    if getId and setId then
        local okG, v = pcall(getId)
        if okG and type(v) == "number" then
            prev = v
            pcall(setId, 8)
        end
    end
    pcall(addRecCore, self, m, args, result)
    if prev then pcall(setId, prev) end
end

local busyCo = nil
local nc = newcclosure or function(f) return f end

if hookmetamethod and getnamecallmethod then
    local old
    old = hookmetamethod(game, "__namecall", nc(function(self, ...)
        local m = getnamecallmethod()
        if recording and not ownCo[coroutine.running()]
            and (m == "FireServer" or m == "InvokeServer")
            and typeof(self) == "Instance" then
            local args = table.pack(...)
            busyCo = coroutine.running()
            if m == "InvokeServer" then
                local res = table.pack(old(self, ...))
                busyCo = nil
                addRec(self, m, args, res)
                return table.unpack(res, 1, res.n)
            end
            addRec(self, m, args)
            local r = table.pack(old(self, ...))
            busyCo = nil
            return table.unpack(r, 1, r.n)
        end
        return old(self, ...)
    end))
    hooked = true
end

-- lớp hook thứ hai: bắt cả các lệnh gọi không đi qua __namecall
if hookfunction then
    pcall(function()
        local ev, fn = Instance.new("RemoteEvent"), Instance.new("RemoteFunction")
        local oldFire, oldInvoke
        oldFire = hookfunction(ev.FireServer, nc(function(self, ...)
            if recording and not ownCo[coroutine.running()] and busyCo ~= coroutine.running()
                and typeof(self) == "Instance" then
                addRec(self, "FireServer", table.pack(...))
            end
            return oldFire(self, ...)
        end))
        oldInvoke = hookfunction(fn.InvokeServer, nc(function(self, ...)
            if recording and not ownCo[coroutine.running()] and busyCo ~= coroutine.running()
                and typeof(self) == "Instance" then
                local args = table.pack(...)
                local res = table.pack(oldInvoke(self, ...))
                addRec(self, "InvokeServer", args, res)
                return table.unpack(res, 1, res.n)
            end
            return oldInvoke(self, ...)
        end))
        hooked = true
    end)
end

    local function findCard()
        local pg = LP:FindFirstChild("PlayerGui")
        if not pg then return end
        for _, o in ipairs(pg:GetDescendants()) do
            if o:IsA("TextLabel") and o.Text ~= "" and #o.Text < 30
                and cleanTxt(o.Text) == "abilities" and isShown(o) then

                while a and a:IsA("GuiObject") and n < 3 do
                    local sz = a.AbsoluteSize
                    if sz.X > 400 or sz.Y > 220 then break end
                    card = a
                    a = a.Parent
                    n += 1
                end
                return o, card
            end
        end
    end

    -- phát lại: bấm vào đúng vị trí tương đối trong khung Abilities
    remoteCache["\0ui"] = function(fx, fy)
        local _, card = findCard()
        if not card then return false, "không thấy khung Abilities" end
        local ap, sz = card.AbsolutePosition, card.AbsoluteSize
        local px, py = ap.X + sz.X * (fx or 0.5), ap.Y + sz.Y * (fy or 0.5)
        local pg = LP:FindFirstChild("PlayerGui")
        local target
        if pg then
            for _, o in ipairs(pg:GetGuiObjectsAtPosition(px, py)) do
                local b = o
                while b and b ~= pg and not b:IsA("GuiButton") do b = b.Parent end
                if b and b:IsA("GuiButton") and isShown(b) then target = b break end
            end
        end
        if target then
            press(target)
            return true
        end
        if VIMa then
            local inset = GSa:GetGuiInset()
            VIMa:SendMouseButtonEvent(px, py + inset.Y, 0, true, game, 0)
            task.wait(0.05)
            VIMa:SendMouseButtonEvent(px, py + inset.Y, 0, false, game, 0)
            return true
        end
        return false, "không bấm được"
    end

    -- ghi: lần bấm vào khung Abilities lúc đang ghi macro (nếu game không có remote Abilities riêng)
    UIS.InputBegan:Connect(function(input)
        if not recording then return end
        local ut = input.UserInputType
        if ut ~= Enum.UserInputType.Touch and ut ~= Enum.UserInputType.MouseButton1 then return end
        local x, y = input.Position.X, input.Position.Y
        local tc, wv, wt = os.clock(), curWave, os.clock() - waveStart
        local ok, _, card = pcall(findCard)
        if not ok or not card then return end
        local ap, sz = card.AbsolutePosition, card.AbsoluteSize
        if x < ap.X or y < ap.Y or x > ap.X + sz.X or y > ap.Y + sz.Y then return end
        local fx, fy = (x - ap.X) / math.max(sz.X, 1), (y - ap.Y) / math.max(sz.Y, 1)

            if not recording then return end

                local e = rec[i]

                local ln = (e.name or ""):lower()
                if ln:find("abilit", 1, true) or ln:find("skill", 1, true) or ln:find("ultimate", 1, true) then
                    return
                end
            end
            if #rec == 0 then recStart = tc end
            local root = rootPart()
            local step = {
                t = tc - recStart, name = "UI:Abilities", method = "UIClick", args = { n = 0 },
                kind = "other", pos = root and root.CFrame or nil,
                wave = wv > 0 and wv or nil, wt = wt, fx = fx, fy = fy,
            }
            local at = #rec + 1
            for i = 1, #rec do
                if rec[i].t > step.t then at = i break end
            end
            table.insert(rec, at, step)
            setStatus("Đã ghi " .. #rec .. " bước · Abilities (bấm giao diện)")
            refreshRec()
        end)
    end)
end

-- ===== MACRO: PHÁT LẠI =====
local runId = 0
local runStarted = 0
local function alive(id) return running and id == runId end

local function short(v)
    if type(v) == "table" then
        return "{" .. tostring(v[1]) .. ", " .. tostring(v[2]) .. "}"
    end
    return tostring(v)
end

-- tìm lại unit (model) trong workspace theo tên + vị trí đã ghi
local function resolveUnit(x)
    local parent = x.parent and workspace:FindFirstChild(x.parent, true)
    local list = parent and parent:GetChildren() or workspace:GetDescendants()
    local best, bd
    for _, m in ipairs(list) do
        if m.Name == x.name and (m:IsA("Model") or m:IsA("BasePart")) then
            local p = posOf(m)
            if p then
                local d = (p - x.pos).Magnitude
                if not bd or d < bd then best, bd = m, d end
            end
        end
    end
    if best and bd < 8 then return best end
end

local function waitForStep(id, e, t0, i, n)
    if S.waveSync and e.wave and curWave > 0 then
        local lastMsg = 0
        while alive(id) and curWave < e.wave do
            if os.clock() - lastMsg > 1 then
                lastMsg = os.clock()
                setStatus(("Vòng %d · bước %d/%d · chờ wave %d (đang wave %d)"):format(n, i, #rec, e.wave, curWave))
            end
            task.wait(0.25)
        end
        if curWave == e.wave then
            while alive(id) do
                local w = (e.wt or 0) - (os.clock() - waveStart)
                if w <= 0 then break end
                setStatus(("Vòng %d · bước %d/%d · chờ %ds trong wave %d"):format(n, i, #rec, math.ceil(w), e.wave))
                task.wait(math.min(w, 0.5))
            end
        end
    else
        while alive(id) do
            local w = e.t - (os.clock() - t0)
            if w <= 0 then break end
            setStatus(("Vòng %d · bước %d/%d · chờ %ds"):format(n, i, #rec, math.ceil(w)))
            task.wait(math.min(w, 0.5))
        end
    end
end

local function playOnce(id, n, skipPast)
    local idMap = {}
    local t0 = os.clock()
    local okN, failN, firstFail = 0, 0, nil
    local nowWave, nowWt = curWave, os.clock() - waveStart
    for i, e in ipairs(rec) do
        if not alive(id) then return okN, failN, firstFail end

        local skip = false
        if skipPast then
            if e.wave and nowWave > 0 then
                skip = e.wave < nowWave or (e.wave == nowWave and (e.wt or 0) <= nowWt + 1)
            else
                skip = true
            end
        end
        if skip then
            if e.placedId ~= nil then idMap[e.placedId] = e.placedId end -- unit đã có sẵn
            logLine("Bước " .. i .. " bỏ qua (đã làm tay trong trận này)")
            continue
        end

        waitForStep(id, e, t0, i, n)
        if not alive(id) then return okN, failN, firstFail end

        -- tới đúng vị trí đã ghi trước khi đặt / nâng / bán
        if S.tpPlace and e.pos and e.kind and e.kind ~= "other" then
            tpTo(e.pos)
        end

        if e.method == "UIClick" then
            local fnU = remoteCache["\0ui"]
            local okU, didU, whyU = ownCall(function() return fnU(e.fx, e.fy) end)
            if okU and didU then
                okN += 1
                setStatus(("Vòng %d · bước %d/%d · Abilities"):format(n, i, #rec))
                logLine("Bước " .. i .. " bấm Abilities")
            else
                failN += 1
                firstFail = firstFail or ("Abilities: " .. tostring(whyU or didU or "lỗi"):sub(1, 60))
                logLine("Bước " .. i .. " Abilities lỗi: " .. tostring(whyU or didU))
            end
            continue
        end

        local inst = e.inst
        if not (inst and inst.Parent) then
            inst = remote(e.name)
            e.inst = inst
        end

        if not inst then
            failN += 1
            firstFail = firstFail or (e.name .. ": không tìm thấy remote")
            logLine("Bước " .. i .. ": không thấy remote " .. e.name)
        else
            local maxTry = (e.placedId ~= nil or e.units) and 6 or 1
            for try = 1, maxTry do
                if not alive(id) then return okN, failN, firstFail end
                local a = table.pack(table.unpack(e.args, 1, e.args.n))
                if e.kind ~= "place" then
                    for ai = 1, a.n do
                        local v = a[ai]
                        if (type(v) == "number" or type(v) == "string") and idMap[v] ~= nil then
                            a[ai] = idMap[v]
                        elseif type(v) == "table" then
                            local c = {}
                            for k, x in pairs(v) do
                                if (type(x) == "number" or type(x) == "string") and idMap[x] ~= nil then
                                    c[k] = idMap[x]
                                else
                                    c[k] = x
                                end
                            end
                            a[ai] = c
                        end
                    end
                elseif (type(a[1]) == "number" or type(a[1]) == "string") and idMap[a[1]] ~= nil then
                    a[1] = idMap[a[1]]
                end

                -- gắn lại unit (model) đã ghi bằng vị trí
                local missing = false
                if e.units then
                    for _, x in ipairs(e.units) do
                        local u = resolveUnit(x)
                        if u then a[x.i] = u else missing = true end
                    end
                end

                if missing then
                    if try == maxTry then
                        failN += 1
                        firstFail = firstFail or (e.name .. ": không tìm thấy unit " .. tostring(e.unit))
                        logLine("Bước " .. i .. ": không thấy unit " .. tostring(e.unit))
                    else
                        task.wait(0.6)
                    end
                else
                    local ok, res = ownCall(function()
                        if e.method == "InvokeServer" then
                            return table.pack(inst:InvokeServer(table.unpack(a, 1, a.n)))
                        end
                        inst:FireServer(table.unpack(a, 1, a.n))
                    end)

                    local r = ok and res and res[1]
                    setStatus(("Vòng %d · bước %d/%d · %s → %s"):format(n, i, #rec, e.name, ok and short(r) or "LỖI"))
                    logLine(("Bước %d %s(%s) → %s"):format(i, e.name, describeArgs(a), ok and short(r) or tostring(res)))

                    if not ok then
                        failN += 1
                        firstFail = firstFail or (e.name .. ": " .. tostring(res):sub(1, 60))
                        break
                    end
                    if e.placedId == nil then
                        okN += 1
                        break
                    end
                    local newId = extractId(res)
                    if newId ~= nil then
                        idMap[e.placedId] = newId
                        okN += 1
                        break
                    end
                    if try == maxTry then
                        failN += 1
                        firstFail = firstFail or (e.name .. " bị từ chối: " .. short(r))
                    else
                        task.wait(1.5)
                    end
                end
            end
        end
    end
    return okN, failN, firstFail
end

local function startMacro(skipPast)
    if #rec == 0 then setStatus("Chưa có macro, hãy bấm Ghi macro trước") return end
    runId += 1
    local id = runId
    running = true
    runStarted = os.clock()
    refreshRun()
    local serial0 = matchSerial
    task.spawn(function()
        local n = 0
        repeat
            n += 1
            local catchUp = skipPast and n == 1
            local okN, failN, firstFail = playOnce(id, n, catchUp)
            if alive(id) then
                setStatus(("Vòng %d xong: ok %d · lỗi %d%s"):format(n, okN, failN,
                    firstFail and (" · " .. firstFail) or ""))
                if S.macroLoop then
                    task.wait(S.cycle)
                    -- Mỗi trận chỉ chạy macro một lần: làm xong thì đợi sang trận mới rồi mới lặp,
                    -- không chạy lại từ đầu ngay trong trận đang chơi (tránh đặt lại unit đã bán)
                    local lastMsg = 0
                    while alive(id) and matchSerial == serial0 do
                        if os.clock() - lastMsg > 1 then
                            lastMsg = os.clock()
                            setStatus(("Vòng %d xong, chờ trận mới để chạy tiếp"):format(n))
                        end
                        task.wait(0.5)
                    end
                    if alive(id) then
                        serial0 = matchSerial
                        task.wait(S.startDelay)
                    end
                end
            end
        until not alive(id) or not S.macroLoop
        if id == runId then
            running = false
            refreshRun()
        end
    end)
end

local function stopMacro()
    runId += 1
    running = false
    setStatus("Đã dừng")
    refreshRun()
end

local function toggleRecord()
    if not hooked then
        setStatus("Executor không hỗ trợ hook, không ghi được")
        return
    end
    recording = not recording
    if recording then
        stopMacro()
        rec, recStart = {}, os.clock()
        placedInfo = {}
        seen, seenOrder, lastShown = {}, {}, ""
        allLog = {}
        setStatus("Đang ghi... hãy đặt / nâng / bán unit")
    else
        if #rec == 0 then
            setStatus("Chưa ghi được bước nào. Bật 'Ghi mọi remote' rồi thử lại")
        else
            saveMacroFile()
            if S.runAfterRec then
                setStatus("Đã ghi xong và lưu, đang chạy macro")
                startMacro(true)
            else
                setStatus("Đã ghi xong và lưu, bấm Chạy macro")
            end
        end
    end
    refreshRec()
end

local function toggleRun()
    if recording then
        recording = false
        saveMacroFile()
        refreshRec()
        startMacro(true)
        return
    end
    if running then stopMacro() else startMacro() end
end

-- Trận mới bắt đầu: tự chạy lại macro (chỉ khi phát hiện đổi trận, không phải đổi wave)
local function onNewMatch()
    if not S.macroAuto or #rec == 0 or recording then return end
    if running and os.clock() - runStarted < 60 then return end -- đừng cắt macro vừa bắt đầu
    local serial = matchSerial
    runId += 1
    running = false
    refreshRun()
    setStatus("Trận mới: chạy macro sau " .. S.startDelay .. " giây")
    task.delay(S.startDelay, function()
        if serial == matchSerial and not recording then startMacro() end
    end)
end

local function trim(t) return (t:gsub("^%s+", ""):gsub("%s+$", "")) end

-- chuẩn hóa chữ: bỏ khoảng trắng đầu cuối, dấu câu ở cuối (ví dụ "Start Game!"), không phân biệt hoa thường
local function norm(t) return (trim(t):lower():gsub("[%p%s]+$", "")) end

-- tìm nhãn / nút đang hiện có đúng chữ; ưu tiên cái nằm trong một nút bấm thật
local function findLabel(want)
    local pg = LP:FindFirstChild("PlayerGui")
    if not pg then return end
    want = norm(want)
    local fallback
    for _, o in ipairs(pg:GetDescendants()) do
        if (o:IsA("TextLabel") or o:IsA("TextButton")) and o.Text ~= "" and #o.Text < 40
            and norm(o.Text) == want and isShown(o) then
            local b = o
            while b and b ~= pg and not b:IsA("GuiButton") do b = b.Parent end
            if b and b:IsA("GuiButton") then return o end
            fallback = fallback or o
        end
    end
    return fallback
end

local function clickAt(o)
    if not VIM then return false end
    return pcall(function()
        local p = o.AbsolutePosition + o.AbsoluteSize / 2
        local inset = GuiService:GetGuiInset()
        VIM:SendMouseButtonEvent(p.X, p.Y + inset.Y, 0, true, game, 0)
        task.wait(0.05)
        VIM:SendMouseButtonEvent(p.X, p.Y + inset.Y, 0, false, game, 0)
    end)
end

-- lần lẻ: bắn tín hiệu nút; lần chẵn: bấm giả vào đúng vị trí (phòng khi game chỉ nhận chạm thật)
local function activate(o, attempt)
    local pg = LP:FindFirstChild("PlayerGui")
    local b = o
    while b and b ~= pg and not b:IsA("GuiButton") do b = b.Parent end
    local isBtn = b and b:IsA("GuiButton")
    if isBtn and attempt % 2 == 1 then
        press(b)
        return true
    end
    return clickAt(isBtn and b or o)
end

-- các bục "Pick a map!" trong sảnh: { pos, n (số người tối đa), text }
local function findPads()
    local pads, seenGui = {}, {}
    local function scan(root)
        for _, o in ipairs(root:GetDescendants()) do
            if o:IsA("TextLabel") and #o.Text < 30 and o.Text:lower():find("pick a map", 1, true) then
                local g = o:FindFirstAncestorWhichIsA("LayerCollector")
                if g and not seenGui[g] then
                    seenGui[g] = true
                    local okA, ad = pcall(function() return g.Adornee end)
                    local part = (okA and ad) or (g.Parent and (g.Parent:IsA("BasePart") or g.Parent:IsA("Model")) and g.Parent)
                    local pos = part and posOf(part)
                    local n
                    for _, d in ipairs(g:GetDescendants()) do
                        if d:IsA("TextLabel") then
                            local mx = trim(d.Text):match("^%d+%s*/%s*(%d+)$")
                            if mx then n = tonumber(mx) break end
                        end
                    end
                    if pos then pads[#pads + 1] = { pos = pos, n = n, part = part } end
                end
            end
        end
    end
    pcall(scan, workspace)
    local pg = LP:FindFirstChild("PlayerGui")
    if pg then pcall(scan, pg) end
    return pads
end

local function choosePad()
    local root = rootPart()
    if not root then return end
    local best, bd, bestAny, bdAny
    for _, pd in ipairs(findPads()) do
        local d = (pd.pos - root.Position).Magnitude
        if not bdAny or d < bdAny then bestAny, bdAny = pd, d end
        if pd.n == S.lobbyPlayers and (not bd or d < bd) then best, bd = pd, d end
    end
    return best or bestAny
end

local lobbyAttempt, lobbyCool, padToggle = 0, 0, false
local function inLobby() return (os.clock() - lastPresent) > 8 end

local function lobbyStep()
    if not S.autoLobby or recording or not inLobby() then return end
    if os.clock() < lobbyCool or not rootPart() then return end
    lobbyAttempt += 1
    local a = lobbyAttempt

    -- 1) màn chọn map đang mở
    local create = findLabel("create lobby")
    if create then
        for _, want in ipairs({ S.mapName, S.mapMode, "Level " .. S.mapLevel }) do
            local lbl = findLabel(want)
            if lbl then
                activate(lbl, a)
                task.wait(0.5)
            end
        end
        activate(create, a)
        setStatus(("Chọn %s · %s · Level %d, đã bấm Create Lobby"):format(S.mapName, S.mapMode, S.mapLevel))
        lobbyCool = os.clock() + 6
        return
    end

    -- 2) hộp "Pick Map"
    local pick = findLabel("pick map")
    if pick then
        activate(pick, a)
        setStatus("Bấm Pick Map")
        return
    end

    -- 3) nút bắt đầu (nếu game cần bấm sau khi tạo phòng)
    for _, w in ipairs({ "start game", "start" }) do
        local st = findLabel(w)
        if st then
            activate(st, a)
            setStatus("Bấm " .. w)
            lobbyCool = os.clock() + 4
            return
        end
    end

    -- 4) đi tới bục chọn map
    local pad = choosePad()
    if pad then
        padToggle = not padToggle
        -- lần sau bước ra rồi bước vào lại để chắc chắn kích hoạt bục
        tpTo(CFrame.new(pad.pos + (padToggle and Vector3.new(0, 3, 0) or Vector3.new(0, 3, 10))))
        setStatus(("Tới bục chọn map (%s người)"):format(tostring(pad.n or "?")))
    else
        setStatus("Không thấy bục 'Pick a map!' trong sảnh")
    end
end

local function drag(handle, target)
    local dragging, moved = false, false
    local start, startPos
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging, moved = true, false
            start, startPos = i.Position, target.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
            or i.UserInputType == Enum.UserInputType.Touch) then
            local d = i.Position - start
            if d.Magnitude > 6 then moved = true end
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
    return function() return moved end
end

local function slider(T, text, min, max, init, cb)
    local key = pendingKey
    pendingKey = nil
    local f = row(T, text, 54)
    local box = new("TextBox", {
        Size = UDim2.new(0, 70, 0, 24), Position = UDim2.new(1, -82, 0, 7),
        BackgroundColor3 = Color3.fromRGB(20, 20, 23),
        Text = tostring(init), TextColor3 = C.txt, TextSize = 13, Font = Enum.Font.GothamMedium,
        ClearTextOnFocus = false,
    }, f)
    corner(box, 5)
    local track = new("Frame", {
        Size = UDim2.new(1, -24, 0, 6), Position = UDim2.new(0, 12, 1, -14),
        BackgroundColor3 = Color3.fromRGB(55, 55, 60), BorderSizePixel = 0,
    }, f)
    corner(track, 3)
    local fill = new("Frame", {
        Size = UDim2.new((init - min) / (max - min), 0, 1, 0), BackgroundColor3 = C.acc, BorderSizePixel = 0,
    }, track)
    corner(fill, 3)

    local function setVal(v)
        v = math.clamp(math.floor(v + 0.5), min, max)
        fill.Size = UDim2.new((v - min) / (max - min), 0, 1, 0)
        box.Text = tostring(v)
        cb(v)
    end
    if key then setters[key] = setVal end
    local dragging = false
    local function upd(i)
        local rel = (i.Position.X - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1)
        setVal(min + math.clamp(rel, 0, 1) * (max - min))
    end
    track.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            upd(i)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
            or i.UserInputType == Enum.UserInputType.Touch) then
            upd(i)
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    box.FocusLost:Connect(function() setVal(tonumber(box.Text) or init) end)
end

local function statLines()
    local p, u, sl, o = 0, 0, 0, 0
    local per, ord = {}, {}
    for _, e in ipairs(rec) do
        local k = e.kind or kindOf(e.name)
        if k == "place" then p += 1
        elseif k == "upgrade" then u += 1
        elseif k == "sell" then sl += 1
        else o += 1 end
        if e.unit and k ~= "other" then
            local s = per[e.unit]
            if not s then
                s = { place = 0, upgrade = 0, sell = 0 }
                per[e.unit] = s
                ord[#ord + 1] = e.unit
            end
            s[k] += 1
        end
    end
    local total = ("Bản ghi: đặt %d · nâng %d · bán %d · khác %d"):format(p, u, sl, o)
    local lines = {}
    for i, name in ipairs(ord) do
        if i > 4 then
            lines[#lines + 1] = "... và " .. (#ord - 4) .. " unit nữa"
            break
        end
        local s = per[name]
        lines[#lines + 1] = ("• %s: đặt %d / nâng %d / bán %d"):format(name, s.place, s.upgrade, s.sell)
    end
    return total, (#lines > 0 and table.concat(lines, "\n") or "Chưa có thống kê theo unit")
end

-- ===== VÒNG LẶP CHÍNH (quét giao diện 1 lần / giây) =====
task.spawn(function()
    local t = 0
    local lastSerial = matchSerial
    while gui.Parent do
        task.wait(1)
        t += 1

        local ok, s = pcall(takeSnapshot)
        if ok and s then
            snap = s
            updateWave(s)
        end
        if matchSerial ~= lastSerial then
            lastSerial = matchSerial
            onNewMatch()
        end

        if S.autoSkip then fire("SetAutoSkip", true) end
        if S.autoSpeed and t % 3 == 0 then fire("ChangeTickSpeed", S.speed) end
        if S.autoReplay then clickSnap("Play Again") end
        if S.autoLeave then clickSnap("Return to lobby") end
        if S.voteLevel > 0 then clickSnap("Level " .. S.voteLevel) end
        if S.upgradeAll and t % 2 == 0 then clickSnap("Upgrade all") end

        if t % 10 == 0 then
            pcall(function() saveFile(FILE_CFG, HttpService:JSONEncode(S)) end)
        end

        if S.noEffects and t % 5 == 0 then
            for _, o in ipairs(workspace:GetDescendants()) do
                if o:IsA("ParticleEmitter") or o:IsA("Trail") or o:IsA("Beam")
                    or o:IsA("Smoke") or o:IsA("Fire") or o:IsA("Sparkles") then
                    o.Enabled = false
                end
            end
        end
        if S.hidePlayers and t % 2 == 0 then
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LP and p.Character then
                    for _, d in ipairs(p.Character:GetDescendants()) do
                        if d:IsA("BasePart") or d:IsA("Decal") then d.LocalTransparencyModifier = 1 end
                    end
                end
            end
        end
    end
end)

RunService.Heartbeat:Connect(function(dt)
    local ch = LP.Character
    local h = ch and ch:FindFirstChildOfClass("Humanoid")
    if h then
        if S.useWalk then h.WalkSpeed = S.walk end
        if S.useJump then
            h.UseJumpPower = true
            h.JumpPower = S.jump
        end
    end
    if S.followMobs and followTarget and os.clock() > holdUntil then
        local r = rootPart()
        local p = r and posOf(followTarget)
        if r and p then
            local flat = Vector3.new(r.Position.X - p.X, 0, r.Position.Z - p.Z)
            local dir = flat.Magnitude > 0.1 and flat.Unit or Vector3.new(0, 0, 1)
            local goal = CFrame.new(p + dir * S.followDist + Vector3.new(0, 3, 0))
            if S.followMode == 1 then
                if (r.Position - goal.Position).Magnitude > 2 then r.CFrame = goal end
            else
                r.CFrame = r.CFrame:Lerp(goal, 1 - math.exp(-dt * 25))
            end
            r.AssemblyLinearVelocity = Vector3.zero
        end
    end
end)
